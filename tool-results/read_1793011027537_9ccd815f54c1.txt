   139→   139→function linesOf(text: string): string[] {
   140→   140→  return text

   153→   153→function hasCondition(r: {
   154→   154→  domain: string[];
   155→   155→  ip: string[];
   156→   156→  port: string;
   157→   157→  network: string;
   158→   158→  protocol: string[];
   159→   159→  inboundTag: string[];
   160→   160→  source: string[];
   161→   161→}): boolean {
   162→   162→  return Boolean(
   163→   163→    r.domain.length ||
   164→   164→      r.ip.length ||
   165→   165→      r.port ||
   166→   166→      r.network ||
   167→   167→      r.protocol.length ||
   168→   168→      r.inboundTag.length ||
   169→   169→      r.source.length
   170→   170→  );
   171→   171→}

   173→   173→function listChip(values: string[]): { text: string; title: string } {
   174→   174→  const title = values.join(" · ");
   175→   175→  const first = values[0] ?? "";
   176→   176→  const head = first.length > 16 ? `${first.slice(0, 15)}…` : first;
   177→   177→  const text = values.length > 1 ? `${head} +${values.length - 1}` : head;
   178→   178→  return { text, title };
   179→   179→}

   181→   181→function netLabel(network: string, u: Dict["routing"]): string {
   182→   182→  if (network === "tcp") return u.netTcp;
   183→   183→  if (network === "udp") return u.netUdp;
   184→   184→  if (network === "tcp,udp" || network === "udp,tcp") return u.netBoth;
   185→   185→  return network;
   186→   186→}

   200→   200→function chipsOf(rule: RoutingRuleDTO, u: Dict["routing"]): CondChip[] {
   201→   201→  const chips: CondChip[] = [];
   202→   202→  if (rule.domain.length) {
   203→   203→    const c = listChip(rule.domain);
   204→   204→    chips.push({ icon: Globe, text: c.text, title: c.title });
   205→   205→  }
   206→   206→  if (rule.ip.length) {
   207→   207→    const c = listChip(rule.ip);
   208→   208→    chips.push({ icon: MapPin, text: c.text, title: c.title });
   209→   209→  }
   210→   210→  if (rule.port) {
   211→   211→    chips.push({ icon: Hash, text: rule.port, title: rule.port });
   212→   212→  }
   213→   213→  if (rule.network) {
   214→   214→    chips.push({ icon: Network, text: netLabel(rule.network, u), title: rule.network });
   215→   215→  }
   216→   216→  if (rule.protocol.length) {
   217→   217→    const c = listChip(rule.protocol);
   218→   218→    chips.push({ icon: ScanSearch, text: c.text, title: c.title });
   219→   219→  }
   220→   220→  if (rule.inboundTag.length) {
   221→   221→    const c = listChip(rule.inboundTag);
   222→   222→    chips.push({ icon: RadioTower, text: c.text, title: c.title });
   223→   223→  }
   224→   224→  if (rule.source.length) {
   225→   225→    const c = listChip(rule.source);
   226→   226→    chips.push({ icon: ArrowUpLeft, text: c.text, title: c.title });
   227→   227→  }
   228→   228→  return chips;
   229→   229→}

   247→   247→  const lang = useAppStore((s) => s.lang);

   253→   253→  const [rules, setRules] = useState<RoutingRuleDTO[] | null>(null);
   254→   254→  const [loadErr, setLoadErr] = useState(false);
   255→   255→
   256→   256→  /* selects source data */
   257→   257→  const [inboundTags, setInboundTags] = useState<string[]>([]);

   272→   272→
   273→   273→  const markDirty = useCallback(() => {
   274→   274→    setDirty(true);
   275→   275→    setChanges((c) => c + 1);
   276→   276→  }, []);
   277→   277→

   284→   284→      setStrategy(
   285→   285→        (STRATEGIES as readonly string[]).includes(d.domainStrategy)
   286→   286→          ? (d.domainStrategy as Strategy)
   287→   287→          : "AsIs"
   288→   288→      );
   289→   289→      setRules(d.rules);
   290→   290→      setDirty(false);
   291→   291→      setChanges(0);
   292→   292→    } catch {

   294→   294→      setLoadErr(true);
   295→   295→      return;

   297→   297→    /* auxiliary selects — non fatal if they fail */

   301→   301→        const d = (await ir.json()) as { inbounds?: { tag?: string }[] };
   302→   302→        setInboundTags(
   303→   303→          (d.inbounds ?? [])
   304→   304→            .map((x) => (typeof x.tag === "string" ? x.tag : ""))
   305→   305→            .filter(Boolean)
   306→   306→        );
   307→   307→      }
   308→   308→    } catch {
   309→   309→      setInboundTags([]);
   310→   310→    }
   311→   311→    try {
   312→   312→      const or = await fetch("/api/outbounds", { cache: "no-store" });
   313→   313→      if (or.ok) {
   314→   314→        const d = (await or.json()) as { outbounds?: { tag?: string }[] };
   315→   315→        setOutboundTags(
   316→   316→          (d.outbounds ?? [])
   317→   317→            .map((x) => (typeof x.tag === "string" ? x.tag : ""))
   318→   318→            .filter(Boolean)
   319→   319→        );
   320→   320→      }
   321→   321→    } catch {
   322→   322→      setOutboundTags([]);
   323→   323→    }
   324→   324→  }, []);
   325→   325→
   326→   326→  useEffect(() => {
   327→   327→    void load();
   328→   328→  }, [load]);
   329→   329→
   330→   330→  /* ---------- derived ---------- */
   331→   331→
   332→   332→  const kpis = useMemo(() => {
   333→   333→    const list = rules ?? [];
   334→   334→    return {
   335→   335→      total: list.length,
   336→   336→      active: list.filter((r) => r.enabled).length,
   337→   337→      blocked: list.filter((r) => r.outboundTag === "blocked").length,
   338→   338→      direct: list.filter((r) => r.outboundTag === "direct").length,

   340→   340→  }, [rules]);
   341→   341→
   342→   342→  /* ---------- list mutations (local, dirty) ---------- */
   343→   343→
   344→   344→  const toggleEnabled = (i: number, enabled: boolean) => {

   346→   346→      if (!prev) return prev;
   347→   347→      const next = [...prev];

   356→   356→      if (!prev) return prev;
   357→   357→      const j = i + dir;

   359→   359→      const next = [...prev];
   360→   360→      const tmp = next[i];
   361→   361→      next[i] = next[j];
   362→   362→      next[j] = tmp;
   363→   363→      return next;
   364→   364→    });

   366→   366→  };

   401→   401→  const applyRule = () => {
   402→   402→    const list = rules ?? [];
   403→   403→    if (!form.outboundTag) {
   404→   404→      toast({ variant: "destructive", description: u.errNeedOutbound });

   409→   409→    const source = csvOf(form.source);
   410→   410→    const port = form.port.trim();
   411→   411→    if (!hasCondition({ domain, ip, port, network: form.network, protocol: form.protocols, inboundTag: form.inboundTags, source })) {
   412→   412→      toast({ variant: "destructive", description: u.errNeedCondition });
   413→   413→      return;

   417→   417→        editIndex !== null && list[editIndex]

   420→   420→      enabled: form.enabled,
   421→   421→      comment: form.comment.trim().slice(0, 80),
   422→   422→      outboundTag: form.outboundTag,
   423→   423→      domain,
   424→   424→      ip,
   425→   425→      port,
   426→   426→      network: form.network,
   427→   427→      protocol: [...form.protocols],
   428→   428→      inboundTag: [...form.inboundTags],
   429→   429→      source,
   430→   430→      sortOrder: editIndex !== null && list[editIndex] ? list[editIndex].sortOrder : list.length,
   431→   431→    };
   432→   432→    setRules((prev) => {
   433→   433→      if (!prev) return prev;
   434→   434→      const next = [...prev];
   435→   435→      if (editIndex !== null && next[editIndex]) next[editIndex] = rule;
   436→   436→      else next.push(rule);
   437→   437→      return next;
   438→   438→    });
   439→   439→    setDlgOpen(false);
   440→   440→    markDirty();
   441→   441→    if (editIndex === null) toast({ description: u.created });
   442→   442→  };
   443→   443→
   444→   444→  /* ---------- bulk save ---------- */
   445→   445→
   446→   446→  const save = async () => {
   447→   447→    const list = rules ?? [];
   448→   448→    for (let i = 0; i < list.length; i++) {
   449→   449→      const r = list[i];
   450→   450→      if (!r.outboundTag) {
   451→   451→        toast({ variant: "destructive", description: `#${i + 1} · ${u.errNeedOutbound}` });
   452→   452→        return;
   453→   453→      }
   454→   454→      if (!hasCondition(r)) {
   455→   455→        toast({ variant: "destructive", description: `#${i + 1} · ${u.errNeedCondition}` });
   456→   456→        return;
   457→   457→      }
   458→   458→    }
   459→   459→    setSaving(true);
   460→   460→    try {
   461→   461→      const res = await fetch("/api/routing", {
   462→   462→        method: "PUT",
   463→   463→        headers: { "Content-Type": "application/json" },
   464→   464→        body: JSON.stringify({
   465→   465→          domainStrategy: strategy,
   466→   466→          rules: list.map((r) => ({
   467→   467→            enabled: r.enabled,
   468→   468→            comment: r.comment,
   469→   469→            outboundTag: r.outboundTag,
   470→   470→            domain: r.domain,
   471→   471→            ip: r.ip,
   472→   472→            port: r.port,
   473→   473→            network: r.network,
   474→   474→            protocol: r.protocol,
   475→   475→            inboundTag: r.inboundTag,
   476→   476→            source: r.source,
   477→   477→          })),
   478→   478→        }),
   479→   479→      });
   480→   480→      if (!res.ok) {
   481→   481→        toast({ variant: "destructive", description: t.inbounds.errGeneric });
   482→   482→        setSaving(false);
   483→   483→        return;
   484→   484→      }
   485→   485→      const d = (await res.json()) as RoutingData;
   486→   486→      setStrategy(
   487→   487→        (STRATEGIES as readonly string[]).includes(d.domainStrategy)
   488→   488→          ? (d.domainStrategy as Strategy)
   489→   489→          : "AsIs"
   490→   490→      );
   491→   491→      setRules(d.rules);
   492→   492→      setDirty(false);
   493→   493→      setChanges(0);
   494→   494→      toast({ description: u.saved });
   495→   495→    } catch {
   496→   496→      toast({ variant: "destructive", description: t.inbounds.errGeneric });
   497→   497→    }
   498→   498→    setSaving(false);
   499→   499→  };
   500→   500→
   501→   501→  const loading = rules === null;
   502→   502→
   503→   503→  /* ---------- render ---------- */
   504→   504→
   505→   505→  if (loading) {
   506→   506→    return (
