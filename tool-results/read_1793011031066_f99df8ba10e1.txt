   166→export function LoginScene({ onSuccess }: { onSuccess: (username: string) => void }) {
   167→  const lang = useAppStore((s) => s.lang);
   168→  const t = useMemo(() => dictionaries[lang].auth, [lang]);
   169→  const { toast } = useToast();
   170→
   171→  /* login form state */
   172→  const [mode, setMode] = useState<Mode>("login");
   173→  const [username, setUsername] = useState("");
   174→  const [password, setPassword] = useState("");
   175→  const [showPw, setShowPw] = useState(false);
   176→  const [remember, setRemember] = useState(true);
   177→  const [busy, setBusy] = useState(false);
   178→
   179→  /* errors + lockout */
   180→  const [error, setError] = useState<{ msg: string; shakeKey: number } | null>(null);
   181→  const [lockSecs, setLockSecs] = useState<number | null>(null);
   182→
   183→  /* recovery state */
   184→  const [recKey, setRecKey] = useState("");
   185→  const [newPw, setNewPw] = useState("");
   186→  const [recBusy, setRecBusy] = useState(false);
   187→  const [recPw, setRecPw] = useState(false);
   188→
   189→  const errShake = useRef(0);
   190→  const mounted = useRef(true);
   191→  useEffect(() => {
   192→    mounted.current = true;
   193→    return () => {
   194→      mounted.current = false;
   195→    };
   196→  }, []);
   197→
   198→  /* lockout countdown */
   199→  useEffect(() => {
   200→    if (lockSecs === null) return;
   201→    if (lockSecs <= 0) {
   202→      setLockSecs(null);
   203→      return;
   204→    }
   205→    const id = setTimeout(() => setLockSecs((s) => (s === null ? null : s - 1)), 1000);
   206→    return () => clearTimeout(id);
   207→  }, [lockSecs]);
   208→
   209→  const fail = (msg: string) =>
   210→    setError({ msg, shakeKey: ++errShake.current });
   211→
   212→  const fmt = (s: number) =>
   213→    `${Math.floor(s / 60)}:${String(s % 60).padStart(2, "0")}`;
   214→
   215→  /* ---------- actions ---------- */
   216→
   217→  async function handleLogin(e: React.FormEvent) {
   218→    e.preventDefault();
   219→    if (busy || lockSecs !== null) return;
   220→    if (!username.trim() || !password) {
   221→      fail(t.errFields);
   222→      return;
   223→    }
   224→    setBusy(true);
   225→    setError(null);
   226→    try {
   227→      const res = await fetch("/api/auth/login", {
   228→        method: "POST",
   229→        headers: { "Content-Type": "application/json" },
   230→        body: JSON.stringify({
   231→          username: username.trim(),
   232→          password,
   233→          remember,
   234→        }),
   235→      });
   236→      const data = (await res.json()) as {
   237→        ok?: boolean;
   238→        username?: string;
   239→        error?: string;
   240→        attemptsLeft?: number;
   241→        retryAfter?: number;
   242→      };
   243→      if (!mounted.current) return;
   244→      if (res.ok && data.ok && data.username) {
   245→        toast({
   246→          title: t.welcomeToast.replace("{name}", data.username),
   247→          description: t.secureNote,
   248→        });
   249→        onSuccess(data.username);
   250→        return;
   251→      }
   252→      if (res.status === 423 && data.retryAfter) {
   253→        setLockSecs(data.retryAfter);
   254→        fail(t.errLocked.replace("{s}", fmt(data.retryAfter)));
   255→        return;
   256→      }
   257→      if (res.status === 401) {
   258→        if (typeof data.attemptsLeft === "number") {
   259→          fail(
   260→            `${t.errInvalid} ${t.attemptsLeft.replace(
   261→              "{n}",
   262→              String(data.attemptsLeft)
   263→            )}`
   264→          );
   265→          return;
   266→        }
   267→        fail(t.errInvalid);
   268→        return;
   269→      }
   270→      fail(t.errFields);
   271→    } catch {
   272→      if (mounted.current) fail(t.errInvalid);
   273→    } finally {
   274→      if (mounted.current) setBusy(false);
   275→    }
   276→  }
   277→
   278→  async function handleRecover(e: React.FormEvent) {
   279→    e.preventDefault();
   280→    if (recBusy || lockSecs !== null) return;
   281→    if (!recKey.trim() || !newPw) {
   282→      fail(t.errFields);
   283→      return;
   284→    }
   285→    if (newPw.length < 8) {
   286→      fail(t.errWeak);
   287→      return;
   288→    }
   289→    setRecBusy(true);
   290→    setError(null);
   291→    try {
   292→      const res = await fetch("/api/auth/recover", {
   293→        method: "POST",
   294→        headers: { "Content-Type": "application/json" },
   295→        body: JSON.stringify({ recoveryKey: recKey.trim(), newPassword: newPw }),
   296→      });
   297→      const data = (await res.json()) as {
   298→        ok?: boolean;
   299→        error?: string;
   300→        attemptsLeft?: number;
   301→        retryAfter?: number;
   302→      };
   303→      if (!mounted.current) return;
   304→      if (res.ok && data.ok) {
   305→        toast({ title: t.okRecovered });
   306→        setMode("login");
   307→        setRecKey("");
   308→        setNewPw("");
   309→        setPassword("");
   310→        setError(null);
   311→        return;
   312→      }
   313→      if (res.status === 423 && data.retryAfter) {
   314→        setLockSecs(data.retryAfter);
   315→        fail(t.errLocked.replace("{s}", fmt(data.retryAfter)));
   316→        return;
   317→      }
   318→      if (data.error === "weak") {
   319→        fail(t.errWeak);
   320→        return;
   321→      }
   322→      if (res.status === 401) {
   323→        if (typeof data.attemptsLeft === "number") {
   324→          fail(
   325→            `${t.errBadKey} ${t.attemptsLeft.replace(
   326→              "{n}",
   327→              String(data.attemptsLeft)
   328→            )}`
   329→          );
   330→          return;
   331→        }
   332→        fail(t.errBadKey);
   333→        return;
   334→      }
   335→      fail(t.errFields);
   336→    } catch {
   337→      if (mounted.current) fail(t.errBadKey);
   338→    } finally {
   339→      if (mounted.current) setRecBusy(false);
   340→    }
   341→  }
   342→
   343→  const autofill = () => {
   344→    setUsername("admin");
   345→    setPassword("sephiroth");
   346→    setError(null);
   347→  };
   348→
   349→  const mmss = lockSecs !== null ? fmt(lockSecs) : "";
   350→
   351→  const lockBanner = lockSecs !== null && (
   352→    <div
   353→      role="alert"
   354→      className="mt-6 flex items-center gap-2.5 rounded-xl border border-amber-400/30 bg-amber-400/10 px-3.5 py-3 text-[13px] text-amber-300"
   355→    >
   356→      <ShieldAlert className="h-4 w-4 shrink-0" />
   357→      <span>{t.errLocked.replace("{s}", mmss)}</span>
   358→      <span className="kbd ms-auto border-amber-400/30 bg-amber-400/10 text-amber-200">
   359→        {mmss}
   360→      </span>
   361→    </div>
   362→  );
   363→
   364→  const errorBanner =
   365→    error && lockSecs === null ? (
   366→      <div
   367→        key={error.shakeKey}
   368→        role="alert"
   369→        className="shake mt-6 flex items-start gap-2.5 rounded-xl border border-destructive/40 bg-destructive/10 px-3.5 py-3 text-[13px] text-destructive-foreground"
   370→      >
   371→        <ShieldAlert className="mt-0.5 h-4 w-4 shrink-0 text-destructive" />
   372→        <span>{error.msg}</span>
   373→      </div>
   374→    ) : null;
   375→
   376→  /* ---------- render ---------- */
   377→
   378→  return (
   379→    <div className="bg-cosmic fixed inset-0 z-50 overflow-y-auto">
   380→      <Starfield />
   381→      <div className="login-horizon" aria-hidden="true" />
   382→
   383→      {/* floating controls */}
   384→      <header className="absolute inset-x-0 top-0 z-20 flex items-center justify-end px-5 py-4 md:px-8">
   385→        <LangMenu />
   386→      </header>
   387→
   388→      <main className="relative z-10 mx-auto flex min-h-full w-full max-w-[1120px] flex-col items-center justify-center gap-5 px-4 py-8 md:px-8 lg:py-12">
   389→        {/* ── framed showcase ── */}
   390→        <div className="glass-frame fade-up relative w-full rounded-[2.1rem] p-2 md:p-2.5">
   391→          <div className="grid overflow-hidden rounded-[1.6rem] border border-white/[.06] bg-background-deep/70 lg:grid-cols-[1.04fr_0.96fr]">
   392→            <ShowcasePanel />
   393→
   394→            {/* ── form side ── */}
   395→            <div className="flex flex-col justify-center px-6 py-9 sm:px-10 md:px-12 md:py-12">
   396→              {mode === "login" ? (
   397→                <div key="login">
   398→                  {/* head */}
   399→                  <div className="glass glow-cyan flex h-14 w-14 items-center justify-center rounded-2xl">
   400→                    <BrandMark size={38} />
   401→                  </div>
   402→                  <h2 className="mt-5 text-[1.7rem] font-bold leading-tight tracking-tight text-foreground">
   403→                    {t.welcome}
   404→                  </h2>
   405→                  <p className="mt-1.5 text-sm text-muted-foreground">
   406→                    {t.welcomeSub}
   407→                  </p>
   408→
   409→                  {lockBanner}
   410→                  {errorBanner}
   411→
   412→                  {/* form */}
   413→                  <form onSubmit={handleLogin} className="mt-7 space-y-5">
   414→                    <div className="space-y-2">
   415→                      <Label htmlFor="login-username" className="text-[13px] font-medium">
   416→                        {t.username}
   417→                      </Label>
   418→                      <div className="relative">
   419→                        <User className="pointer-events-none absolute start-3.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   420→                        <Input
   421→                          id="login-username"
   422→                          autoComplete="username"
   423→                          value={username}
   424→                          onChange={(e) => setUsername(e.target.value)}
   425→                          placeholder={t.usernamePh}
   426→                          disabled={busy || lockSecs !== null}
   427→                          className={fieldCls}
   428→                        />
   429→                      </div>
   430→                    </div>
   431→
   432→                    <div className="space-y-2">
   433→                      <div className="flex items-center justify-between gap-3">
   434→                        <Label htmlFor="login-password" className="text-[13px] font-medium">
   435→                          {t.password}
   436→                        </Label>
   437→                        <button
   438→                          type="button"
   439→                          onClick={() => {
   440→                            setMode("recover");
   441→                            setError(null);
   442→                          }}
   443→                          className="text-xs font-medium text-primary underline-offset-4 outline-none transition-colors hover:underline focus-visible:ring-2 focus-visible:ring-ring"
   444→                        >
   445→                          {t.forgotShort}
   446→                        </button>
   447→                      </div>
   448→                      <div className="relative">
   449→                        <KeyRound className="pointer-events-none absolute start-3.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   450→                        <Input
   451→                          id="login-password"
   452→                          type={showPw ? "text" : "password"}
   453→                          autoComplete="current-password"
   454→                          value={password}
   455→                          onChange={(e) => setPassword(e.target.value)}
   456→                          placeholder={t.passwordPh}
   457→                          disabled={busy || lockSecs !== null}
   458→                          className={cn(fieldCls, "pe-11")}
   459→                        />
   460→                        <button
   461→                          type="button"
   462→                          onClick={() => setShowPw((v) => !v)}
   463→                          aria-label={showPw ? "Hide password" : "Show password"}
   464→                          className="absolute end-3 top-1/2 -translate-y-1/2 rounded-md p-1 text-muted-foreground outline-none transition-colors hover:text-foreground focus-visible:ring-2 focus-visible:ring-ring"
   465→                        >
   466→                          {showPw ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   467→                        </button>
   468→                      </div>
   469→                    </div>
   470→
   471→                    <label
   472→                      htmlFor="login-remember"
   473→                      className="flex cursor-pointer items-center gap-2.5 text-[13px] text-muted-foreground"
   474→                    >
   475→                      <Checkbox
   476→                        id="login-remember"
   477→                        checked={remember}
   478→                        onCheckedChange={(v) => setRemember(v === true)}
   479→                        disabled={busy || lockSecs !== null}
   480→                        className="border-border/70"
   481→                      />
   482→                      {t.remember}
   483→                    </label>
   484→
   485→                    <Button
   486→                      type="submit"
   487→                      disabled={busy || lockSecs !== null}
   488→                      className="relative mt-1 h-12 w-full overflow-hidden rounded-xl bg-primary text-[15px] font-semibold text-primary-foreground shadow-[0_0_26px_-8px_rgba(34,216,238,0.6)] transition-all hover:bg-primary/90 hover:shadow-[0_0_34px_-6px_rgba(34,216,238,0.7)] disabled:opacity-60"
   489→                    >
   490→                      {busy ? (
   491→                        <>
   492→                          <Loader2 className="me-2 h-4 w-4 animate-spin" />
   493→                          {t.signingIn}
   494→                        </>
   495→                      ) : (
   496→                        <>
   497→                          <Lock className="me-2 h-3.5 w-3.5 opacity-70" />
   498→                          {t.signIn}
   499→                        </>
   500→                      )}
   501→                      {!busy && lockSecs === null && (
   502→                        <span
   503→                          aria-hidden="true"
   504→                          className="shine-sweep pointer-events-none absolute inset-y-0 w-1/3 bg-gradient-to-r from-transparent via-white/20 to-transparent"
   505→                        />
   506→                      )}
   507→                    </Button>
   508→                  </form>
   509→
   510→                  {/* or divider */}
   511→                  <div className="mt-6">
   512→                    <OrDivider label={t.or} />
   513→                  </div>
   514→
   515→                  {/* utility row (in place of social logins) */}
   516→                  <div className="mt-6 grid grid-cols-2 gap-3">
   517→                    <Button
   518→                      type="button"
   519→                      variant="outline"
   520→                      onClick={() => {
   521→                        setMode("recover");
   522→                        setError(null);
   523→                      }}
   524→                      disabled={lockSecs !== null}
   525→                      className="glass hover-glow h-11 rounded-xl border-border/60 bg-transparent text-[13px] font-medium text-foreground"
   526→                    >
   527→                      <ShieldCheck className="me-2 h-4 w-4 text-primary" />
   528→                      {t.useRecovery}
   529→                    </Button>
   530→                    <Button
   531→                      type="button"
   532→                      variant="outline"
   533→                      onClick={autofill}
   534→                      disabled={lockSecs !== null}
   535→                      className="glass hover-glow h-11 rounded-xl border-border/60 bg-transparent text-[13px] font-medium text-foreground"
   536→                    >
   537→                      <WandSparkles className="me-2 h-4 w-4 text-primary" />
   538→                      {t.demoFill}
   539→                    </Button>
   540→                  </div>
   541→
   542→                  {/* preview credentials */}
   543→                  <p className="mt-5 text-center text-[11px] text-muted-foreground/85">
   544→                    {t.demoLabel}
   545→                    <span className="mx-1.5 text-border">·</span>
   546→                    <span className="font-mono text-foreground/80">admin / sephiroth</span>
   547→                  </p>
   548→                </div>
   549→              ) : (
   550→                /* ── recovery mode ── */
