   139→function linesOf(text: string): string[] {
   140→  return text
   141→    .split("\n")
   142→    .map((s) => s.trim())
   143→    .filter(Boolean);
   144→}
   145→
   146→function csvOf(text: string): string[] {
   147→  return text
   148→    .split(",")
   149→    .map((s) => s.trim())
   150→    .filter(Boolean);
   151→}
   152→
   153→function hasCondition(r: {
   154→  domain: string[];
   155→  ip: string[];
   156→  port: string;
   157→  network: string;
   158→  protocol: string[];
   159→  inboundTag: string[];
   160→  source: string[];
   161→}): boolean {
   162→  return Boolean(
   163→    r.domain.length ||
   164→      r.ip.length ||
   165→      r.port ||
   166→      r.network ||
   167→      r.protocol.length ||
   168→      r.inboundTag.length ||
   169→      r.source.length
   170→  );
   171→}
   172→
   173→function listChip(values: string[]): { text: string; title: string } {
   174→  const title = values.join(" · ");
   175→  const first = values[0] ?? "";
   176→  const head = first.length > 16 ? `${first.slice(0, 15)}…` : first;
   177→  const text = values.length > 1 ? `${head} +${values.length - 1}` : head;
   178→  return { text, title };
   179→}
   180→
   181→function netLabel(network: string, u: Dict["routing"]): string {
   182→  if (network === "tcp") return u.netTcp;
   183→  if (network === "udp") return u.netUdp;
   184→  if (network === "tcp,udp" || network === "udp,tcp") return u.netBoth;
   185→  return network;
   186→}

   188→function obChipCls(tag: string): string {
   189→  if (tag === "direct") return "ring-chart-3/40 bg-chart-3/10 text-chart-3";
   190→  if (tag === "blocked") return "ring-chart-5/40 bg-chart-5/10 text-chart-5";
   191→  return "ring-primary/40 bg-primary/10 text-primary";
   192→}
   193→
   194→function obDotCls(tag: string): string {
   195→  if (tag === "direct") return "bg-chart-3";
   196→  if (tag === "blocked") return "bg-chart-5";
   197→  return "bg-primary";
   198→}

   200→function chipsOf(rule: RoutingRuleDTO, u: Dict["routing"]): CondChip[] {
   201→  const chips: CondChip[] = [];
   202→  if (rule.domain.length) {
   203→    const c = listChip(rule.domain);
   204→    chips.push({ icon: Globe, text: c.text, title: c.title });
   205→  }
   206→  if (rule.ip.length) {
   207→    const c = listChip(rule.ip);
   208→    chips.push({ icon: MapPin, text: c.text, title: c.title });
   209→  }
   210→  if (rule.port) {
   211→    chips.push({ icon: Hash, text: rule.port, title: rule.port });
   212→  }
   213→  if (rule.network) {
   214→    chips.push({ icon: Network, text: netLabel(rule.network, u), title: rule.network });
   215→  }
   216→  if (rule.protocol.length) {
   217→    const c = listChip(rule.protocol);
   218→    chips.push({ icon: ScanSearch, text: c.text, title: c.title });
   219→  }
   220→  if (rule.inboundTag.length) {
   221→    const c = listChip(rule.inboundTag);
   222→    chips.push({ icon: RadioTower, text: c.text, title: c.title });
   223→  }
   224→  if (rule.source.length) {
   225→    const c = listChip(rule.source);
   226→    chips.push({ icon: ArrowUpLeft, text: c.text, title: c.title });
   227→  }
   228→  return chips;
   229→}

   231→function toForm(r: RoutingRuleDTO): RuleForm {
   232→  return {
   233→    enabled: r.enabled,
   234→    comment: r.comment,
   235→    outboundTag: r.outboundTag,
   236→    domainText: r.domain.join("\n"),
   237→    ipText: r.ip.join("\n"),
   238→    port: r.port,
   239→    network: r.network,
   240→    protocols: [...r.protocol],
   241→    inboundTags: [...r.inboundTag],
   242→    source: r.source.join(", "),
   243→  };
   244→}

   246→export function RoutingView() {
   247→  const lang = useAppStore((s) => s.lang);
   248→  const t = useMemo(() => dictionaries[lang], [lang]);
   249→  const u = t.routing;
   250→  const { toast } = useToast();
   251→
   252→  const [strategy, setStrategy] = useState<Strategy>("AsIs");
   253→  const [rules, setRules] = useState<RoutingRuleDTO[] | null>(null);
   254→  const [loadErr, setLoadErr] = useState(false);
   255→
   256→  /* selects source data */
   257→  const [inboundTags, setInboundTags] = useState<string[]>([]);
   258→  const [outboundTags, setOutboundTags] = useState<string[]>([]);
   259→
   260→  /* dirty tracking */
   261→  const [dirty, setDirty] = useState(false);
   262→  const [changes, setChanges] = useState(0);
   263→  const [saving, setSaving] = useState(false);
   264→
   265→  /* dialog */
   266→  const [dlgOpen, setDlgOpen] = useState(false);
   267→  const [editIndex, setEditIndex] = useState<number | null>(null);
   268→  const [form, setForm] = useState<RuleForm>(EMPTY_RULE);
   269→
   270→  /* delete */
   271→  const [delIndex, setDelIndex] = useState<number | null>(null);
   272→
   273→  const markDirty = useCallback(() => {
   274→    setDirty(true);
   275→    setChanges((c) => c + 1);
   276→  }, []);
   277→
   278→  const load = useCallback(async () => {
   279→    setLoadErr(false);
   280→    try {
   281→      const res = await fetch("/api/routing", { cache: "no-store" });
   282→      if (!res.ok) throw new Error("routing");
   283→      const d = (await res.json()) as RoutingData;
   284→      setStrategy(
   285→        (STRATEGIES as readonly string[]).includes(d.domainStrategy)
   286→          ? (d.domainStrategy as Strategy)
   287→          : "AsIs"
   288→      );
   289→      setRules(d.rules);
   290→      setDirty(false);
   291→      setChanges(0);
   292→    } catch {
   293→      setRules([]);
   294→      setLoadErr(true);
   295→      return;
   296→    }
   297→    /* auxiliary selects — non fatal if they fail */
   298→    try {
   299→      const ir = await fetch("/api/inbounds", { cache: "no-store" });
   300→      if (ir.ok) {
   301→        const d = (await ir.json()) as { inbounds?: { tag?: string }[] };
   302→        setInboundTags(
   303→          (d.inbounds ?? [])
   304→            .map((x) => (typeof x.tag === "string" ? x.tag : ""))
   305→            .filter(Boolean)
   306→        );
   307→      }
   308→    } catch {
   309→      setInboundTags([]);
   310→    }
   311→    try {
   312→      const or = await fetch("/api/outbounds", { cache: "no-store" });
   313→      if (or.ok) {
   314→        const d = (await or.json()) as { outbounds?: { tag?: string }[] };
   315→        setOutboundTags(
   316→          (d.outbounds ?? [])
   317→            .map((x) => (typeof x.tag === "string" ? x.tag : ""))
   318→            .filter(Boolean)
   319→        );
   320→      }
   321→    } catch {
   322→      setOutboundTags([]);
   323→    }
   324→  }, []);
   325→
   326→  useEffect(() => {
   327→    void load();
   328→  }, [load]);
   329→
   330→  /* ---------- derived ---------- */
   331→
   332→  const kpis = useMemo(() => {
   333→    const list = rules ?? [];
   334→    return {
   335→      total: list.length,
   336→      active: list.filter((r) => r.enabled).length,
   337→      blocked: list.filter((r) => r.outboundTag === "blocked").length,
   338→      direct: list.filter((r) => r.outboundTag === "direct").length,
   339→    };
   340→  }, [rules]);
   341→
   342→  /* ---------- list mutations (local, dirty) ---------- */
   343→
   344→  const toggleEnabled = (i: number, enabled: boolean) => {
   345→    setRules((prev) => {
   346→      if (!prev) return prev;
   347→      const next = [...prev];
   348→      next[i] = { ...next[i], enabled };
   349→      return next;
   350→    });
   351→    markDirty();
   352→  };
   353→
   354→  const move = (i: number, dir: -1 | 1) => {
   355→    setRules((prev) => {
   356→      if (!prev) return prev;
   357→      const j = i + dir;
   358→      if (j < 0 || j >= prev.length) return prev;
   359→      const next = [...prev];
   360→      const tmp = next[i];
   361→      next[i] = next[j];
   362→      next[j] = tmp;
   363→      return next;
   364→    });
   365→    markDirty();
   366→  };
   367→
   368→  const removeRule = () => {
   369→    if (delIndex === null) return;
   370→    setRules((prev) => (prev ? prev.filter((_, i) => i !== delIndex) : prev));
   371→    setDelIndex(null);
   372→    markDirty();
   373→    toast({ description: u.deleted });
   374→  };
   375→
   376→  /* ---------- rule dialog ---------- */
   377→
   378→  const openCreate = () => {
   379→    setEditIndex(null);
   380→    setForm(EMPTY_RULE);
   381→    setDlgOpen(true);
   382→  };
   383→
   384→  const openEdit = (i: number) => {
   385→    const list = rules ?? [];
   386→    if (!list[i]) return;
   387→    setEditIndex(i);
   388→    setForm(toForm(list[i]));
   389→    setDlgOpen(true);
   390→  };
   391→
   392→  const toggleIn = (key: "protocols" | "inboundTags", value: string, on: boolean) => {
   393→    setForm((f) => {
   394→      const set = new Set(f[key]);
   395→      if (on) set.add(value);
   396→      else set.delete(value);
   397→      return { ...f, [key]: [...set] };
   398→    });
   399→  };
   400→
   401→  const applyRule = () => {
   402→    const list = rules ?? [];
   403→    if (!form.outboundTag) {
   404→      toast({ variant: "destructive", description: u.errNeedOutbound });
   405→      return;
   406→    }
   407→    const domain = linesOf(form.domainText);
   408→    const ip = linesOf(form.ipText);
   409→    const source = csvOf(form.source);
   410→    const port = form.port.trim();
   411→    if (!hasCondition({ domain, ip, port, network: form.network, protocol: form.protocols, inboundTag: form.inboundTags, source })) {
   412→      toast({ variant: "destructive", description: u.errNeedCondition });
   413→      return;
   414→    }
   415→    const rule: RoutingRuleDTO = {
   416→      id:
   417→        editIndex !== null && list[editIndex]
   418→          ? list[editIndex].id
   419→          : `tmp-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 8)}`,
   420→      enabled: form.enabled,
   421→      comment: form.comment.trim().slice(0, 80),
   422→      outboundTag: form.outboundTag,
   423→      domain,
   424→      ip,
   425→      port,
   426→      network: form.network,
   427→      protocol: [...form.protocols],
   428→      inboundTag: [...form.inboundTags],
   429→      source,
   430→      sortOrder: editIndex !== null && list[editIndex] ? list[editIndex].sortOrder : list.length,
   431→    };
   432→    setRules((prev) => {
   433→      if (!prev) return prev;
   434→      const next = [...prev];
   435→      if (editIndex !== null && next[editIndex]) next[editIndex] = rule;
   436→      else next.push(rule);
   437→      return next;
   438→    });
   439→    setDlgOpen(false);
   440→    markDirty();
   441→    if (editIndex === null) toast({ description: u.created });
   442→  };
   443→
   444→  /* ---------- bulk save ---------- */
   445→
   446→  const save = async () => {
   447→    const list = rules ?? [];
   448→    for (let i = 0; i < list.length; i++) {
   449→      const r = list[i];
   450→      if (!r.outboundTag) {
   451→        toast({ variant: "destructive", description: `#${i + 1} · ${u.errNeedOutbound}` });
   452→        return;
   453→      }
   454→      if (!hasCondition(r)) {
   455→        toast({ variant: "destructive", description: `#${i + 1} · ${u.errNeedCondition}` });
   456→        return;
   457→      }
   458→    }
   459→    setSaving(true);
   460→    try {
   461→      const res = await fetch("/api/routing", {
   462→        method: "PUT",
   463→        headers: { "Content-Type": "application/json" },
   464→        body: JSON.stringify({
   465→          domainStrategy: strategy,
   466→          rules: list.map((r) => ({
   467→            enabled: r.enabled,
   468→            comment: r.comment,
   469→            outboundTag: r.outboundTag,
   470→            domain: r.domain,
   471→            ip: r.ip,
   472→            port: r.port,
   473→            network: r.network,
   474→            protocol: r.protocol,
   475→            inboundTag: r.inboundTag,
   476→            source: r.source,
   477→          })),
   478→        }),
   479→      });
   480→      if (!res.ok) {
   481→        toast({ variant: "destructive", description: t.inbounds.errGeneric });
   482→        setSaving(false);
   483→        return;
   484→      }
   485→      const d = (await res.json()) as RoutingData;
   486→      setStrategy(
   487→        (STRATEGIES as readonly string[]).includes(d.domainStrategy)
   488→          ? (d.domainStrategy as Strategy)
   489→          : "AsIs"
   490→      );
   491→      setRules(d.rules);
   492→      setDirty(false);
   493→      setChanges(0);
   494→      toast({ description: u.saved });
   495→    } catch {
   496→      toast({ variant: "destructive", description: t.inbounds.errGeneric });
   497→    }
   498→    setSaving(false);
   499→  };
   500→
   501→  const loading = rules === null;
   502→
   503→  /* ---------- render ---------- */
   504→
   505→  if (loading) {
   506→    return (
   507→      <div className="fade-up space-y-6">
   508→        <div className="space-y-2">
   509→          <Skeleton className="h-6 w-28 rounded-full" />
   510→          <Skeleton className="h-9 w-52" />
   511→          <Skeleton className="h-4 w-80" />
   512→        </div>
   513→        <Skeleton className="h-28 w-full rounded-2xl" />
   514→        <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-4">
   515→          {Array.from({ length: 4 }).map((_, i) => (
   516→            <Skeleton key={i} className="h-24 rounded-2xl" />
   517→          ))}
   518→        </div>
   519→        <div className="space-y-3">
   520→          {Array.from({ length: 2 }).map((_, i) => (
   521→            <Skeleton key={i} className="h-24 rounded-2xl" />
   522→          ))}
   523→        </div>
   524→      </div>
   525→    );
   526→  }
   527→
   528→  if (loadErr) {
   529→    return (
   530→      <div className="fade-up space-y-6">
   531→        <div
   532→          className="glass flex items-center gap-3 rounded-2xl border border-chart-5/30 px-5 py-4 text-sm text-chart-5"
   533→          role="alert"
   534→        >
   535→          <TriangleAlert className="h-4 w-4 shrink-0" aria-hidden="true" />
   536→          {t.inbounds.errGeneric}
   537→          <Button
   538→            variant="outline"
   539→            size="sm"
   540→            onClick={() => void load()}
   541→            className="ms-auto gap-2 rounded-full border-white/10 bg-white/[0.03]"
   542→            aria-label="retry"
   543→          >
   544→            <RotateCw className="h-4 w-4" aria-hidden="true" />
   545→          </Button>
   546→        </div>
   547→      </div>
   548→    );
   549→  }
   550→
   551→  return (
   552→    <div className="fade-up space-y-6">
   553→      {/* header */}
   554→      <section className="space-y-2">
   555→        <span className="glass inline-flex items-center gap-1.5 rounded-full px-3 py-1 text-[10px] font-semibold tracking-[0.22em] text-primary/90 ring-1 ring-primary/20">
   556→          <Route className="h-3 w-3" aria-hidden="true" />
   557→          {u.badge}
   558→        </span>
   559→        <div className="flex flex-wrap items-end justify-between gap-3">
   560→          <div>
   561→            <h2 className="text-silver text-3xl font-bold tracking-tight md:text-4xl">
   562→              {t.nav.routing}
   563→            </h2>
   564→            <p className="mt-1 max-w-2xl text-sm text-muted-foreground">{u.subtitle}</p>
   565→          </div>
   566→          <Button onClick={openCreate} className="glow-cyan gap-2" aria-label={u.new}>
   567→            <Plus className="h-4 w-4" aria-hidden="true" />
   568→            {u.new}
   569→          </Button>
   570→        </div>
   571→      </section>
   572→
   573→      {/* tip banner */}
   574→      <div className="glass max-w-2xl rounded-2xl">
   575→        <p className="flex items-start gap-2 px-4 py-3 text-sm">
   576→          <Info className="mt-0.5 h-4 w-4 shrink-0 text-primary" aria-hidden="true" />
   577→          <span>
   578→            <span className="font-medium text-foreground">{u.tipTitle}</span>
   579→            <span className="mt-0.5 block text-xs leading-relaxed text-muted-foreground">
   580→              {u.tipBody}
   581→            </span>
   582→          </span>
   583→        </p>
   584→      </div>
   585→
   586→      {/* strategy card */}
   587→      <section className="glass rounded-2xl p-5 md:p-6" aria-label={u.strategy}>
   588→        <div className="mb-4 flex items-start gap-3">
   589→          <span className="glass flex h-9 w-9 shrink-0 items-center justify-center rounded-xl">
   590→            <Network className="h-4 w-4 text-primary" aria-hidden="true" />
   591→          </span>
   592→          <div className="min-w-0">
   593→            <h3 className="text-sm font-semibold text-foreground">{u.strategy}</h3>
   594→            <p className="mt-0.5 text-xs text-muted-foreground">{u.strategyDesc}</p>
   595→          </div>
   596→        </div>
   597→        <Select
   598→          value={strategy}
   599→          onValueChange={(v) => {
   600→            setStrategy(v as Strategy);
   601→            markDirty();
