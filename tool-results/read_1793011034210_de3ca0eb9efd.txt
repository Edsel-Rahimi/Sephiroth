   166→   166→export function LoginScene({ onSuccess }: { onSuccess: (username: string) => void }) {
   167→   167→  const lang = useAppStore((s) => s.lang);
   168→   168→  const t = useMemo(() => dictionaries[lang].auth, [lang]);
   169→   169→  const { toast } = useToast();
   170→   170→
   171→   171→  /* login form state */
   172→   172→  const [mode, setMode] = useState<Mode>("login");
   173→   173→  const [username, setUsername] = useState("");
   174→   174→  const [password, setPassword] = useState("");
   175→   175→  const [showPw, setShowPw] = useState(false);
   176→   176→  const [remember, setRemember] = useState(true);
   177→   177→  const [busy, setBusy] = useState(false);
   178→   178→
   179→   179→  /* errors + lockout */
   180→   180→  const [error, setError] = useState<{ msg: string; shakeKey: number } | null>(null);
   181→   181→  const [lockSecs, setLockSecs] = useState<number | null>(null);
   182→   182→
   183→   183→  /* recovery state */
   184→   184→  const [recKey, setRecKey] = useState("");
   185→   185→  const [newPw, setNewPw] = useState("");
   186→   186→  const [recBusy, setRecBusy] = useState(false);
   187→   187→  const [recPw, setRecPw] = useState(false);
   188→   188→
   189→   189→  const errShake = useRef(0);
   190→   190→  const mounted = useRef(true);
   191→   191→  useEffect(() => {
   192→   192→    mounted.current = true;
   193→   193→    return () => {
   194→   194→      mounted.current = false;
   195→   195→    };
   196→   196→  }, []);
   197→   197→
   198→   198→  /* lockout countdown */
   199→   199→  useEffect(() => {
   200→   200→    if (lockSecs === null) return;
   201→   201→    if (lockSecs <= 0) {
   202→   202→      setLockSecs(null);
   203→   203→      return;
   204→   204→    }
   205→   205→    const id = setTimeout(() => setLockSecs((s) => (s === null ? null : s - 1)), 1000);
   206→   206→    return () => clearTimeout(id);
   207→   207→  }, [lockSecs]);
   208→   208→
   209→   209→  const fail = (msg: string) =>
   210→   210→    setError({ msg, shakeKey: ++errShake.current });
   211→   211→
   212→   212→  const fmt = (s: number) =>
   213→   213→    `${Math.floor(s / 60)}:${String(s % 60).padStart(2, "0")}`;
   214→   214→
   215→   215→  /* ---------- actions ---------- */
   216→   216→
   217→   217→  async function handleLogin(e: React.FormEvent) {
   218→   218→    e.preventDefault();
   219→   219→    if (busy || lockSecs !== null) return;
   220→   220→    if (!username.trim() || !password) {
   221→   221→      fail(t.errFields);
   222→   222→      return;
   223→   223→    }
   224→   224→    setBusy(true);
   225→   225→    setError(null);
   226→   226→    try {
   227→   227→      const res = await fetch("/api/auth/login", {
   228→   228→        method: "POST",
   229→   229→        headers: { "Content-Type": "application/json" },
   230→   230→        body: JSON.stringify({
   231→   231→          username: username.trim(),
   232→   232→          password,
   233→   233→          remember,
   234→   234→        }),
   235→   235→      });
   236→   236→      const data = (await res.json()) as {
   237→   237→        ok?: boolean;
   238→   238→        username?: string;
   239→   239→        error?: string;
   240→   240→        attemptsLeft?: number;
   241→   241→        retryAfter?: number;
   242→   242→      };
   243→   243→      if (!mounted.current) return;
   244→   244→      if (res.ok && data.ok && data.username) {
   245→   245→        toast({
   246→   246→          title: t.welcomeToast.replace("{name}", data.username),
   247→   247→          description: t.secureNote,
   248→   248→        });
   249→   249→        onSuccess(data.username);
   250→   250→        return;
   251→   251→      }
   252→   252→      if (res.status === 423 && data.retryAfter) {
   253→   253→        setLockSecs(data.retryAfter);
   254→   254→        fail(t.errLocked.replace("{s}", fmt(data.retryAfter)));
   255→   255→        return;
   256→   256→      }
   257→   257→      if (res.status === 401) {
   258→   258→        if (typeof data.attemptsLeft === "number") {
   259→   259→          fail(
   260→   260→            `${t.errInvalid} ${t.attemptsLeft.replace(
   261→   261→              "{n}",
   262→   262→              String(data.attemptsLeft)
   263→   263→            )}`
   264→   264→          );
   265→   265→          return;
   266→   266→        }
   267→   267→        fail(t.errInvalid);
   268→   268→        return;
   269→   269→      }
   270→   270→      fail(t.errFields);
   271→   271→    } catch {
   272→   272→      if (mounted.current) fail(t.errInvalid);
   273→   273→    } finally {
   274→   274→      if (mounted.current) setBusy(false);
   275→   275→    }
   276→   276→  }
   277→   277→
   278→   278→  async function handleRecover(e: React.FormEvent) {
   279→   279→    e.preventDefault();
   280→   280→    if (recBusy || lockSecs !== null) return;
   281→   281→    if (!recKey.trim() || !newPw) {
   282→   282→      fail(t.errFields);
   283→   283→      return;
   284→   284→    }
   285→   285→    if (newPw.length < 8) {
   286→   286→      fail(t.errWeak);
   287→   287→      return;
   288→   288→    }
   289→   289→    setRecBusy(true);
   290→   290→    setError(null);
   291→   291→    try {
   292→   292→      const res = await fetch("/api/auth/recover", {
   293→   293→        method: "POST",
   294→   294→        headers: { "Content-Type": "application/json" },
   295→   295→        body: JSON.stringify({ recoveryKey: recKey.trim(), newPassword: newPw }),
   296→   296→      });
   297→   297→      const data = (await res.json()) as {
   298→   298→        ok?: boolean;
   299→   299→        error?: string;
   300→   300→        attemptsLeft?: number;
   301→   301→        retryAfter?: number;
   302→   302→      };
   303→   303→      if (!mounted.current) return;
   304→   304→      if (res.ok && data.ok) {
   305→   305→        toast({ title: t.okRecovered });
   306→   306→        setMode("login");
   307→   307→        setRecKey("");
   308→   308→        setNewPw("");
   309→   309→        setPassword("");
   310→   310→        setError(null);
   311→   311→        return;
   312→   312→      }
   313→   313→      if (res.status === 423 && data.retryAfter) {
   314→   314→        setLockSecs(data.retryAfter);
   315→   315→        fail(t.errLocked.replace("{s}", fmt(data.retryAfter)));
   316→   316→        return;
   317→   317→      }
   318→   318→      if (data.error === "weak") {
   319→   319→        fail(t.errWeak);
   320→   320→        return;
   321→   321→      }
   322→   322→      if (res.status === 401) {
   323→   323→        if (typeof data.attemptsLeft === "number") {
   324→   324→          fail(
   325→   325→            `${t.errBadKey} ${t.attemptsLeft.replace(
   326→   326→              "{n}",
   327→   327→              String(data.attemptsLeft)
   328→   328→            )}`
   329→   329→          );
   330→   330→          return;
   331→   331→        }
   332→   332→        fail(t.errBadKey);
   333→   333→        return;
   334→   334→      }
   335→   335→      fail(t.errFields);
   336→   336→    } catch {
   337→   337→      if (mounted.current) fail(t.errBadKey);
   338→   338→    } finally {
   339→   339→      if (mounted.current) setRecBusy(false);
   340→   340→    }
   341→   341→  }
   342→   342→
   343→   343→  const autofill = () => {
   344→   344→    setUsername("admin");
   345→   345→    setPassword("sephiroth");
   346→   346→    setError(null);
   347→   347→  };
   348→   348→
   349→   349→  const mmss = lockSecs !== null ? fmt(lockSecs) : "";
   350→   350→
   351→   351→  const lockBanner = lockSecs !== null && (
   352→   352→    <div
   353→   353→      role="alert"
   354→   354→      className="mt-6 flex items-center gap-2.5 rounded-xl border border-amber-400/30 bg-amber-400/10 px-3.5 py-3 text-[13px] text-amber-300"
   355→   355→    >
   356→   356→      <ShieldAlert className="h-4 w-4 shrink-0" />
   357→   357→      <span>{t.errLocked.replace("{s}", mmss)}</span>
   358→   358→      <span className="kbd ms-auto border-amber-400/30 bg-amber-400/10 text-amber-200">
   359→   359→        {mmss}
   360→   360→      </span>
   361→   361→    </div>
   362→   362→  );
   363→   363→
   364→   364→  const errorBanner =
   365→   365→    error && lockSecs === null ? (
   366→   366→      <div
   367→   367→        key={error.shakeKey}
   368→   368→        role="alert"
   369→   369→        className="shake mt-6 flex items-start gap-2.5 rounded-xl border border-destructive/40 bg-destructive/10 px-3.5 py-3 text-[13px] text-destructive-foreground"
   370→   370→      >
   371→   371→        <ShieldAlert className="mt-0.5 h-4 w-4 shrink-0 text-destructive" />
   372→   372→        <span>{error.msg}</span>
   373→   373→      </div>
   374→   374→    ) : null;
   375→   375→
   376→   376→  /* ---------- render ---------- */
   377→   377→
   378→   378→  return (
   379→   379→    <div className="bg-cosmic fixed inset-0 z-50 overflow-y-auto">
   380→   380→      <Starfield />
   381→   381→      <div className="login-horizon" aria-hidden="true" />
   382→   382→
   383→   383→      {/* floating controls */}
   384→   384→      <header className="absolute inset-x-0 top-0 z-20 flex items-center justify-end px-5 py-4 md:px-8">
   385→   385→        <LangMenu />
   386→   386→      </header>
   387→   387→
   388→   388→      <main className="relative z-10 mx-auto flex min-h-full w-full max-w-[1120px] flex-col items-center justify-center gap-5 px-4 py-8 md:px-8 lg:py-12">
   389→   389→        {/* ── framed showcase ── */}
   390→   390→        <div className="glass-frame fade-up relative w-full rounded-[2.1rem] p-2 md:p-2.5">
   391→   391→          <div className="grid overflow-hidden rounded-[1.6rem] border border-white/[.06] bg-background-deep/70 lg:grid-cols-[1.04fr_0.96fr]">
   392→   392→            <ShowcasePanel />
   393→   393→
   394→   394→            {/* ── form side ── */}
   395→   395→            <div className="flex flex-col justify-center px-6 py-9 sm:px-10 md:px-12 md:py-12">
   396→   396→              {mode === "login" ? (
   397→   397→                <div key="login">
   398→   398→                  {/* head */}
   399→   399→                  <div className="glass glow-cyan flex h-14 w-14 items-center justify-center rounded-2xl">
   400→   400→                    <BrandMark size={38} />
   401→   401→                  </div>
   402→   402→                  <h2 className="mt-5 text-[1.7rem] font-bold leading-tight tracking-tight text-foreground">
   403→   403→                    {t.welcome}
   404→   404→                  </h2>
   405→   405→                  <p className="mt-1.5 text-sm text-muted-foreground">
   406→   406→                    {t.welcomeSub}
   407→   407→                  </p>
   408→   408→
   409→   409→                  {lockBanner}
   410→   410→                  {errorBanner}
   411→   411→
   412→   412→                  {/* form */}
   413→   413→                  <form onSubmit={handleLogin} className="mt-7 space-y-5">
   414→   414→                    <div className="space-y-2">
   415→   415→                      <Label htmlFor="login-username" className="text-[13px] font-medium">
   416→   416→                        {t.username}
   417→   417→                      </Label>
   418→   418→                      <div className="relative">
   419→   419→                        <User className="pointer-events-none absolute start-3.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   420→   420→                        <Input
   421→   421→                          id="login-username"
   422→   422→                          autoComplete="username"
   423→   423→                          value={username}
   424→   424→                          onChange={(e) => setUsername(e.target.value)}
   425→   425→                          placeholder={t.usernamePh}
   426→   426→                          disabled={busy || lockSecs !== null}
   427→   427→                          className={fieldCls}
   428→   428→                        />
   429→   429→                      </div>
   430→   430→                    </div>
   431→   431→
   432→   432→                    <div className="space-y-2">
   433→   433→                      <div className="flex items-center justify-between gap-3">
   434→   434→                        <Label htmlFor="login-password" className="text-[13px] font-medium">
   435→   435→                          {t.password}
   436→   436→                        </Label>
   437→   437→                        <button
   438→   438→                          type="button"
   439→   439→                          onClick={() => {
   440→   440→                            setMode("recover");
   441→   441→                            setError(null);
   442→   442→                          }}
   443→   443→                          className="text-xs font-medium text-primary underline-offset-4 outline-none transition-colors hover:underline focus-visible:ring-2 focus-visible:ring-ring"
   444→   444→                        >
   445→   445→                          {t.forgotShort}
   446→   446→                        </button>
   447→   447→                      </div>
   448→   448→                      <div className="relative">

   450→   450→                        <Input
