    90→function avatarColor(seed: string): string {
    91→  const palette = [
    92→    "bg-primary/20 text-primary",
    93→    "bg-[#8b8cf8]/20 text-[#8b8cf8]",
    94→    "bg-chart-3/20 text-chart-3",
    95→    "bg-chart-4/20 text-chart-4",
    96→    "bg-[#c084fc]/20 text-[#c084fc]",
    97→    "bg-[#f472b6]/20 text-[#f472b6]",
    98→  ];
    99→  let h = 0;
   100→  for (let i = 0; i < seed.length; i++) h = (h * 31 + seed.charCodeAt(i)) >>> 0;
   101→  return palette[h % palette.length];
   102→}

   106→export function SubGroupsView() {
   107→  const lang = useAppStore((s) => s.lang);
   108→  const t = useMemo(() => dictionaries[lang], [lang]);
   109→  const u = t.subgroups;
   110→
   111→  const { toast } = useToast();
   112→
   113→  const [groups, setGroups] = useState<GroupDTO[] | null>(null);
   114→  const [users, setUsers] = useState<UserDTO[]>([]);
   115→  const [loadErr, setLoadErr] = useState(false);
   116→  const [origin, setOrigin] = useState("");
   117→
   118→  /* toolbar */
   119→  const [q, setQ] = useState("");
   120→
   121→  /* dialog */
   122→  const [dlgOpen, setDlgOpen] = useState(false);
   123→  const [editing, setEditing] = useState<GroupDTO | null>(null);
   124→  const [form, setForm] = useState<GrpForm>(EMPTY_FORM);

   126→  const [saving, setSaving] = useState(false);
   127→  const [memberQ, setMemberQ] = useState("");
   128→
   129→  /* delete */
   130→  const [deleting, setDeleting] = useState<GroupDTO | null>(null);
   131→  const [deleteBusy, setDeleteBusy] = useState(false);
   132→
   133→  useEffect(() => {
   134→    setOrigin(window.location.origin);
   135→  }, []);
   136→
   137→  const load = useCallback(async () => {
   138→    try {
   139→      const [gr, ur] = await Promise.all([
   140→        fetch("/api/sub/groups", { cache: "no-store" }),
   141→        fetch("/api/users", { cache: "no-store" }),
   142→      ]);
   143→      if (!gr.ok || !ur.ok) throw new Error("load");
   144→      const gj = (await gr.json()) as { groups: GroupDTO[] };
   145→      const uj = (await ur.json()) as { users: UserDTO[] };
   146→      setGroups(gj.groups);
   147→      setUsers(uj.users);
   148→      setLoadErr(false);
   149→    } catch {
   150→      setGroups([]);
   151→      setLoadErr(true);
   152→    }
   153→  }, []);
   154→
   155→  useEffect(() => {
   156→    void load();
   157→  }, [load]);
   158→
   159→  /* ---------- derived ---------- */
   160→
   161→  const userById = useMemo(() => {
   162→    const m = new Map<string, UserDTO>();
   163→    for (const x of users) m.set(x.id, x);
   164→    return m;
   165→  }, [users]);
   166→
   167→  const kpis = useMemo(() => {
   168→    const list = groups ?? [];
   169→    const members = new Set<string>();
   170→    for (const g of list) for (const id of g.memberIds) members.add(id);
   171→    return {
   172→      groups: list.length,
   173→      members: members.size,
   174→      protected: list.filter((g) => g.hasPassword).length,
   175→    };
   176→  }, [groups]);
   177→
   178→  const filtered = useMemo(() => {
   179→    const needle = q.trim().toLowerCase();
   180→    let list = [...(groups ?? [])];
   181→    if (needle) {
   182→      list = list.filter(
   183→        (g) =>
   184→          g.name.toLowerCase().includes(needle) ||
   185→          g.token.includes(needle) ||
   186→          g.memberIds.some((id) => (userById.get(id)?.username ?? "").toLowerCase().includes(needle))
   187→      );
   188→    }
   189→    list.sort((a, b) => b.createdAt.localeCompare(a.createdAt));
   190→    return list;
   191→  }, [groups, q, userById]);
   192→
   193→  const memberCandidates = useMemo(() => {
   194→    const needle = memberQ.trim().toLowerCase();
   195→    return users.filter((x) => !needle || x.username.toLowerCase().includes(needle));
   196→  }, [users, memberQ]);
   197→
   198→  /* ---------- actions ---------- */
   199→
   200→  const openCreate = () => {
   201→    setEditing(null);
   202→    setForm(EMPTY_FORM);
   203→    setMemberQ("");
   204→    setFieldErr({});
   205→    setDlgOpen(true);
   206→  };
   207→
   208→  const openEdit = (g: GroupDTO) => {
   209→    setEditing(g);
   210→    setForm({
   211→      name: g.name,
   212→      infoLine: g.infoLine ?? "",
   213→      password: "",
   214→      clearPassword: false,
   215→      enabled: g.enabled,
   216→      memberIds: new Set(g.memberIds),
   217→    });
   218→    setMemberQ("");
   219→    setFieldErr({});
   220→    setDlgOpen(true);
   221→  };
   222→
   223→  const toggleMember = (id: string) => {
   224→    setForm((f) => {
   225→      const next = new Set(f.memberIds);
   226→      if (next.has(id)) next.delete(id);
   227→      else next.add(id);
   228→      return { ...f, memberIds: next };
   229→    });
   230→  };
   231→
   232→  const save = async () => {
   233→    setSaving(true);
   234→    setFieldErr({});
   235→    const name = form.name.trim();
   236→    if (name.length < 1 || name.length > 48) {
   237→      setFieldErr({ name: u.errName });
   238→      setSaving(false);
   239→      return;
   240→    }
   241→    if (form.password && form.password.length < 4) {
   242→      setFieldErr({ password: u.pwShort });
   243→      setSaving(false);
   244→      return;
   245→    }
   246→    try {
   247→      const payload: Record<string, unknown> = {
   248→        name,
   249→        infoLine: form.infoLine.trim(),
   250→        enabled: form.enabled,
   251→        memberIds: [...form.memberIds],
   252→      };
   253→      if (form.clearPassword) payload.clearPassword = true;
   254→      else if (form.password) payload.password = form.password;
   255→
   256→      const res = await fetch(editing ? `/api/sub/groups/${editing.id}` : "/api/sub/groups", {
   257→        method: editing ? "PATCH" : "POST",
   258→        headers: { "Content-Type": "application/json" },
   259→        body: JSON.stringify(payload),
   260→      });
   261→      if (res.status === 400) {
   262→        const j = (await res.json()) as { fields?: { field: string; code: string }[] };
   263→        const map: Partial<Record<string, string>> = {};
   264→        for (const f of j.fields ?? []) map[f.field] = f.code === "short" ? u.pwShort : u.errName;
   265→        setFieldErr(map);
   266→        setSaving(false);
   267→        return;
   268→      }
   269→      if (!res.ok) throw new Error("save");
   270→      await load();
   271→      setDlgOpen(false);
   272→      toast({
   273→        description: editing ? u.toastUpdated : `${u.toastCreated} · ${name}`,
   274→      });
   275→    } catch {
   276→      setFieldErr({ name: u.errName });
   277→    }
   278→    setSaving(false);
   279→  };
   280→
   281→  const destroy = async (g: GroupDTO) => {
   282→    setDeleteBusy(true);
   283→    try {
   284→      const res = await fetch(`/api/sub/groups/${g.id}`, { method: "DELETE" });
   285→      if (!res.ok) throw new Error("delete");
   286→      setGroups((prev) => (prev ?? []).filter((x) => x.id !== g.id));
   287→      setDeleting(null);
   288→      toast({ description: `${u.toastDeleted} · ${g.name}` });
   289→    } catch {
   290→      setDeleting(null);
   291→    }
   292→    setDeleteBusy(false);
   293→  };
   294→
   295→  const toggleEnabled = async (g: GroupDTO, enabled: boolean) => {
   296→    setGroups((prev) => (prev ?? []).map((x) => (x.id === g.id ? { ...x, enabled } : x)));
   297→    try {
   298→      await fetch(`/api/sub/groups/${g.id}`, {
   299→        method: "PATCH",
   300→        headers: { "Content-Type": "application/json" },
   301→        body: JSON.stringify({ enabled }),
   302→      });
   303→    } catch {
   304→      setGroups((prev) => (prev ?? []).map((x) => (x.id === g.id ? { ...x, enabled: !enabled } : x)));
   305→    }
   306→  };
   307→
   308→  const copyText = async (text: string, label: string) => {
   309→    try {
   310→      await navigator.clipboard.writeText(text);
   311→    } catch {
   312→      /* noop */
   313→    }
   314→    toast({ description: label });
   315→  };
   316→
   317→  const publicLinkOf = (g: GroupDTO) => `${origin}/sub/${g.token}`;
   318→
   319→  const loading = groups === null;
   320→
   321→  /* ---------- render ---------- */
   322→
   323→  return (
   324→    <div className="fade-up space-y-6">
   325→      {/* header */}
   326→      <section className="space-y-2">
   327→        <span className="glass inline-flex items-center gap-1.5 rounded-full px-3 py-1 text-[10px] font-semibold tracking-[0.22em] text-primary/90 ring-1 ring-primary/20">
   328→          <Layers className="h-3 w-3" aria-hidden="true" />
   329→          {u.badge}
   330→        </span>
   331→        <div className="flex flex-wrap items-end justify-between gap-3">
   332→          <div>
   333→            <h1 className="text-silver text-3xl font-bold tracking-tight md:text-4xl">{t.nav.subgroups}</h1>
   334→            <p className="mt-1 max-w-2xl text-sm text-muted-foreground">{u.subtitle}</p>
   335→          </div>
   336→          <Button onClick={openCreate} className="glow-cyan gap-2" aria-label={u.new}>
   337→            <FolderPlus className="h-4 w-4" aria-hidden="true" />
   338→            {u.new}
   339→          </Button>
   340→        </div>
   341→      </section>
   342→
   343→      {/* KPI row */}
   344→      <section className="grid grid-cols-3 gap-3" aria-label={u.kpiGroups}>
   345→        {[
   346→          { label: u.kpiGroups, value: kpis.groups, icon: Layers, cls: "text-primary" },
   347→          { label: u.kpiMembers, value: kpis.members, icon: UsersIcon, cls: "text-chart-3" },
   348→          { label: u.kpiProtected, value: kpis.protected, icon: KeyRound, cls: "text-chart-4" },
   349→        ].map(({ label, value, icon: Icon, cls }) => (
   350→          <div key={label} className="glass rounded-2xl p-4">
   351→            <div className="flex items-center justify-between gap-2">
   352→              <span className="text-[11px] font-medium text-muted-foreground">{label}</span>
   353→              <span className={cn("glass flex h-7 w-7 items-center justify-center rounded-lg ring-1 ring-white/10", cls)}>
   354→                <Icon className="h-3.5 w-3.5" aria-hidden="true" />
   355→              </span>
   356→            </div>
   357→            <p className="mt-1.5 font-mono text-2xl font-bold text-foreground md:text-3xl">{value}</p>
   358→          </div>
   359→        ))}
   360→      </section>
   361→
   362→      {/* search */}
   363→      <section className="relative max-w-md">
   364→        <Search className="absolute start-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" aria-hidden="true" />
   365→        <Input
   366→          value={q}
   367→          onChange={(e) => setQ(e.target.value)}
   368→          placeholder={u.searchPh}
   369→          className="h-10 ps-9"
   370→          aria-label={u.searchPh}
   371→        />
   372→      </section>
   373→      <Tip>{u.tipGroups}</Tip>
   374→
   375→      {/* list */}
   376→      {loading ? (
   377→        <div className="grid gap-4 md:grid-cols-2">
   378→          {Array.from({ length: 2 }).map((_, i) => (
   379→            <Skeleton key={i} className="h-52 rounded-2xl" />
   380→          ))}
   381→        </div>
   382→      ) : filtered.length === 0 ? (
   383→        <div className="glass rounded-2xl p-12 text-center ring-1 ring-white/10">
   384→          <Layers className="mx-auto h-10 w-10 text-muted-foreground/50" aria-hidden="true" />
   385→          <p className="mt-4 text-sm font-semibold text-foreground">{u.empty}</p>
   386→          <p className="mt-1 text-xs text-muted-foreground">{u.emptyDesc}</p>
   387→        </div>
   388→      ) : (
   389→        <ul className="grid gap-4 md:grid-cols-2">
   390→          {filtered.map((g) => (
   391→            <li
   392→              key={g.id}
   393→              className={cn(
   394→                "glass hover-glow relative overflow-hidden rounded-2xl p-5 ring-1 ring-white/10",
   395→                !g.enabled && "opacity-60"
   396→              )}
   397→            >
   398→              <span className="absolute inset-x-0 top-0 h-0.5 bg-gradient-to-r from-transparent via-primary/50 to-transparent" />
   399→              {/* name + protection */}
   400→              <div className="flex items-start justify-between gap-3">
   401→                <div className="flex min-w-0 items-center gap-2.5">
   402→                  <span
   403→                    className={cn(
   404→                      "flex h-9 w-9 shrink-0 items-center justify-center rounded-xl ring-1 ring-white/10",
   405→                      avatarColor(g.token)
   406→                    )}
   407→                  >
   408→                    <Layers className="h-4 w-4" aria-hidden="true" />
   409→                  </span>
   410→                  <div className="min-w-0">
   411→                    <p className="truncate text-sm font-bold text-foreground">{g.name}</p>
   412→                    <p className="mt-0.5 flex items-center gap-1.5 text-[10px] text-muted-foreground">
   413→                      <CalendarDays className="h-3 w-3" aria-hidden="true" />
   414→                      {u.memberSince} {new Date(g.createdAt).toLocaleDateString(lang)}
   415→                    </p>
   416→                  </div>

   418→                <span
   419→                  className={cn(
   420→                    "inline-flex shrink-0 items-center gap-1 rounded-full px-2 py-0.5 text-[10px] font-semibold ring-1",
   421→                    g.hasPassword

   428→                </span>
   429→              </div>
   430→
   431→              {/* token */}
   432→              <div className="mt-3.5">
   433→                <p className="text-[10px] font-medium uppercase tracking-[0.18em] text-muted-foreground">{u.colToken}</p>
   434→                <div className="mt-1.5 flex items-center gap-2">

   479→                  {g.memberIds.length > 6 && (

   498→                  {u.enabled}
   499→                </label>
   500→                <div className="flex items-center gap-1">
   501→                  <Button
   502→                    variant="outline"
   503→                    size="sm"
   504→                    onClick={() => window.open(publicLinkOf(g), "_blank")}
   505→                    aria-label={u.preview}
