    70→
    71→interface InboundDTO {
    72→  id: string;
    73→  tag: string;
    74→  protocol: string;
    75→  port: number;
    76→  transport: string;
    77→  security: string;
    78→  status: string;
    79→  note: string | null;
    80→  createdAt: string;
    81→  updatedAt: string;
    82→}

    84→type StatusFilter = "all" | "online" | "maintenance" | "disabled";
    85→type FormState = {
    86→  tag: string;
    87→  protocol: string;
    88→  port: string;
    89→  transport: string;

    92→  note: string;
    93→};

    95→const PROTOCOLS = ["vless", "vmess", "trojan", "shadowsocks", "hysteria2"] as const;
    96→const TRANSPORTS = ["tcp", "ws", "grpc", "httpupgrade"] as const;
    97→const STATUSES = ["online", "maintenance", "disabled"] as const;
    98→
    99→const PROTO_LABEL: Record<string, string> = {
   100→  vless: "VLESS",
   101→  vmess: "VMess",
   102→  trojan: "Trojan",
   103→  shadowsocks: "Shadowsocks",

   108→  vless: "bg-primary/10 text-primary ring-primary/30",
   109→  vmess: "bg-chart-2/10 text-chart-2 ring-chart-2/30",
   110→  trojan: "bg-chart-4/10 text-chart-4 ring-chart-4/30",

   115→const SCHEME: Record<string, string> = {
   116→  vless: "vless",
   117→  vmess: "vmess",
   118→  trojan: "trojan",
   119→  shadowsocks: "ss",

   121→};
   122→
   123→const STATUS_DOT: Record<string, string> = {
   124→  online: "bg-chart-3",
   125→  maintenance: "bg-chart-4",
   126→  disabled: "bg-chart-5",

   128→
   129→const TRANSPORT_SHORT: Record<string, string> = {
   130→  tcp: "",
   131→  ws: "WS",
   132→  grpc: "gRPC",
   133→  httpupgrade: "HTTPUpgrade",
   134→};
   135→
   136→/** Same composition the core driver uses for its wire `proto` label. */
   137→function composeProto(protocol: string, transport: string, security: string): string {
   138→  const parts = [PROTO_LABEL[protocol] ?? protocol];
   139→  const tr = TRANSPORT_SHORT[transport] ?? transport.toUpperCase();
   140→  if (tr) parts.push(tr);
   141→  if (security === "tls") parts.push("TLS");
   142→  return parts.join(" · ");
   143→}
   144→
   145→function fmtTraffic(gb: number): string {
   146→  if (gb >= 1000) return `${(gb / 1000).toFixed(2)} TB`;
   147→  return `${Math.round(gb)} GB`;
   148→}

   150→export function InboundsView() {
   151→  const lang = useAppStore((s) => s.lang);
   152→  const t = useMemo(() => dictionaries[lang], [lang]);
   153→  const { toast } = useToast();
   154→
   155→  /* live stats from the core adapter (wire list is a stable ref) */
   156→  const wireInbounds = useCoreStore((s) => s.snapshot.inbounds);
   157→  const upsertInboundLocal = useCoreStore((s) => s.upsertInboundLocal);
   158→  const removeInboundLocal = useCoreStore((s) => s.removeInboundLocal);
   159→
   160→  /* config rows from the API — null = first load */
   161→  const [rows, setRows] = useState<InboundDTO[] | null>(null);
   162→  const [loadErr, setLoadErr] = useState(false);
   163→
   164→  /* toolbar */
   165→  const [q, setQ] = useState("");
   166→  const [statusFilter, setStatusFilter] = useState<StatusFilter>("all");
   167→
   168→  /* create / edit dialog */
   169→  const [dialogOpen, setDialogOpen] = useState(false);
   170→  const [editing, setEditing] = useState<InboundDTO | null>(null);
   171→  const [form, setForm] = useState<FormState>({
   172→    tag: "",
   173→    protocol: "vless",
   174→    port: "",
   175→    transport: "tcp",
   176→    security: "none",
   177→    status: "online",
   178→    note: "",
   179→  });
   180→  const [fieldErr, setFieldErr] = useState<Partial<Record<keyof FormState, string>>>({});
   181→  const [topErr, setTopErr] = useState<string | null>(null);
   182→  const [saving, setSaving] = useState(false);
   183→
   184→  /* delete confirm */
   185→  const [deleting, setDeleting] = useState<InboundDTO | null>(null);
   186→  const [deleteBusy, setDeleteBusy] = useState(false);
   187→
   188→  const FIELD_ERR: Record<string, string> = {
   189→    required: t.inbounds.errGeneric,
   190→    tagFormat: t.inbounds.errTagFormat,
   191→    portType: t.inbounds.errPortType,
   192→    portRange: t.inbounds.errPortRange,
   193→    noteLong: t.inbounds.errNoteLong,
   194→    badValue: t.inbounds.errBadValue,
   195→  };
   196→
   197→  /* ---------- data ---------- */
   198→
   199→  const load = useCallback(async () => {
   200→    try {
   201→      const res = await fetch("/api/inbounds", { cache: "no-store" });
   202→      if (!res.ok) {
   203→        setLoadErr(true);
   204→        return;
   205→      }
   206→      const data = (await res.json()) as { inbounds?: InboundDTO[] };
   207→      setRows(data.inbounds ?? []);
   208→      setLoadErr(false);
   209→    } catch {
   210→      setLoadErr(true);
   211→    }
   212→  }, []);
   213→
   214→  useEffect(() => {
   215→    void load();
   216→  }, [load]);
   217→
   218→  const statsByTag = useMemo(() => {
   219→    const map = new Map<string, { users: number; txGb: number }>();
   220→    for (const ib of wireInbounds) map.set(ib.tag, { users: ib.users, txGb: ib.txGb });
   221→    return map;
   222→  }, [wireInbounds]);
   223→
   224→  const merged = useMemo(
   225→    () =>
   226→      (rows ?? []).map((r) => {
   227→        const live = statsByTag.get(r.tag);
   228→        return { ...r, users: live?.users ?? 0, txGb: live?.txGb ?? 0 };
   229→      }),
   230→    [rows, statsByTag]
   231→  );
   232→
   233→  const counts = useMemo(
   234→    () => ({
   235→      all: merged.length,
   236→      online: merged.filter((r) => r.status === "online").length,
   237→      maintenance: merged.filter((r) => r.status === "maintenance").length,
   238→      disabled: merged.filter((r) => r.status === "disabled").length,
   239→    }),
   240→    [merged]
   241→  );
   242→
   243→  const totals = useMemo(
   244→    () => ({
   245→      users: merged.reduce((acc, r) => acc + r.users, 0),
   246→      traffic: merged.reduce((acc, r) => acc + r.txGb, 0),
   247→    }),
   248→    [merged]
   249→  );
   250→
   251→  const visible = useMemo(() => {
   252→    const needle = q.trim().toLowerCase();
   253→    return merged.filter(
   254→      (r) =>
   255→        (statusFilter === "all" || r.status === statusFilter) &&
   256→        (!needle || r.tag.toLowerCase().includes(needle))
   257→    );
   258→  }, [merged, q, statusFilter]);
   259→
   260→  /* ---------- mutations ---------- */
   261→
   262→  const openCreate = () => {
   263→    const used = new Set((rows ?? []).map((r) => r.port));
   264→    let port = 10000;
   265→    while (used.has(port)) port++;
   266→    setEditing(null);
   267→    setForm({ tag: "", protocol: "vless", port: String(port), transport: "tcp", security: "none", status: "online", note: "" });
   268→    setFieldErr({});
   269→    setTopErr(null);
   270→    setDialogOpen(true);
   271→  };
   272→
   273→  const openEdit = (row: InboundDTO) => {
   274→    setEditing(row);
   275→    setForm({
   276→      tag: row.tag,
   277→      protocol: row.protocol,
   278→      port: String(row.port),
   279→      transport: row.transport,
   280→      security: row.security,
   281→      status: row.status,
   282→      note: row.note ?? "",
   283→    });
   284→    setFieldErr({});
   285→    setTopErr(null);
   286→    setDialogOpen(true);
   287→  };
   288→
   289→  const save = async () => {
   290→    if (saving) return;
   291→    setSaving(true);
   292→    setTopErr(null);
   293→    setFieldErr({});
   294→    const payload: Record<string, unknown> = {
   295→      tag: form.tag.trim(),
   296→      protocol: form.protocol,
   297→      port: form.port.trim() === "" ? "" : Number(form.port),
   298→      transport: form.transport,
   299→      security: form.security,
   300→      status: form.status,
   301→      note: form.note.trim(), // "" normalizes to null server-side
   302→    };
   303→    try {
   304→      const res = await fetch(editing ? `/api/inbounds/${editing.id}` : "/api/inbounds", {
   305→        method: editing ? "PATCH" : "POST",
   306→        headers: { "Content-Type": "application/json" },
   307→        body: JSON.stringify(payload),
   308→      });
   309→      const data = (await res.json().catch(() => ({}))) as {
   310→        inbound?: InboundDTO;
   311→        error?: string;
   312→        fields?: Record<string, string>;
   313→      };
   314→
   315→      if (res.status === 401) {
   316→        setTopErr(t.inbounds.errGeneric);
   317→        return;
   318→      }
   319→      if (res.status === 400 && data.fields) {
   320→        const mapped: Partial<Record<keyof FormState, string>> = {};
   321→        for (const [k, v] of Object.entries(data.fields)) {
   322→          mapped[k as keyof FormState] = FIELD_ERR[v] ?? t.inbounds.errBadValue;
   323→        }
   324→        setFieldErr(mapped);
   325→        return;
   326→      }
   327→      if (res.status === 409) {
   328→        setTopErr(
   329→          data.error === "duplicate_tag"
   330→            ? t.inbounds.errTagUsed
   331→            : data.error === "duplicate_port"
   332→              ? t.inbounds.errPortUsed
   333→              : t.inbounds.errGeneric
   334→        );
   335→        return;
   336→      }
   337→      if (!res.ok || !data.inbound) {
   338→        toast({ title: t.inbounds.toastFail, variant: "destructive" });
   339→        return;
   340→      }
   341→
   342→      const dto = data.inbound;
   343→      const oldTag = editing?.tag;
   344→      if (editing && oldTag && oldTag !== dto.tag) {
   345→        removeInboundLocal(oldTag);
   346→        upsertInboundLocal({
   347→          tag: dto.tag,
   348→          proto: composeProto(dto.protocol, dto.transport, dto.security),
   349→          port: dto.port,
   350→          status: dto.status as "online" | "maintenance" | "disabled",
   351→          users: 0,
   352→          txGb: 0,
   353→        });
   354→      } else {
   355→        const prev = wireInbounds.find((x) => x.tag === (oldTag ?? dto.tag));
   356→        upsertInboundLocal({
   357→          tag: dto.tag,
   358→          proto: composeProto(dto.protocol, dto.transport, dto.security),
   359→          port: dto.port,
   360→          status: dto.status as "online" | "maintenance" | "disabled",
   361→          users: prev?.users ?? 0,
   362→          txGb: prev?.txGb ?? 0,
   363→        });
   364→      }
   365→
   366→      setRows((prev) => {
   367→        const base = prev ?? [];
   368→        return editing
   369→          ? base.map((r) => (r.id === dto.id ? dto : r))
   370→          : [...base, dto];
   371→      });
   372→      setDialogOpen(false);
   373→      toast({
   374→        title: lfmt(editing ? t.inbounds.toastUpdated : t.inbounds.toastCreated, { tag: dto.tag }),
   375→      });
   376→      if (!editing) void load(); // canonical ordering for fresh rows
   377→    } catch {
   378→      toast({ title: t.inbounds.toastFail, variant: "destructive" });
   379→    } finally {
   380→      setSaving(false);
   381→    }
   382→  };
   383→
   384→  const confirmDelete = async () => {
   385→    if (!deleting || deleteBusy) return;
   386→    setDeleteBusy(true);
   387→    try {
   388→      const res = await fetch(`/api/inbounds/${deleting.id}`, { method: "DELETE" });
   389→      if (res.ok || res.status === 404) {
   390→        removeInboundLocal(deleting.tag);
   391→        setRows((prev) => (prev ?? []).filter((r) => r.id !== deleting.id));
   392→        toast({ title: lfmt(t.inbounds.toastDeleted, { tag: deleting.tag }) });
   393→        setDeleting(null);
   394→      } else {
   395→        toast({ title: t.inbounds.toastFail, variant: "destructive" });
   396→      }
   397→    } catch {
   398→      toast({ title: t.inbounds.toastFail, variant: "destructive" });
   399→    } finally {
   400→      setDeleteBusy(false);
   401→    }
   402→  };
   403→
   404→  const setStatus = async (row: InboundDTO, status: string) => {
   405→    try {
   406→      const res = await fetch(`/api/inbounds/${row.id}`, {
   407→        method: "PATCH",
   408→        headers: { "Content-Type": "application/json" },
   409→        body: JSON.stringify({ status }),
   410→      });
   411→      const data = (await res.json().catch(() => ({}))) as { inbound?: InboundDTO };
   412→      if (!res.ok || !data.inbound) {
   413→        toast({ title: t.inbounds.toastFail, variant: "destructive" });
   414→        return;
   415→      }
   416→      upsertInboundLocal({
   417→        tag: row.tag,
   418→        proto: composeProto(row.protocol, row.transport, row.security),
   419→        port: row.port,
   420→        status: status as "online" | "maintenance" | "disabled",
   421→        users: statsByTag.get(row.tag)?.users ?? 0,
   422→        txGb: statsByTag.get(row.tag)?.txGb ?? 0,
   423→      });
   424→      setRows((prev) => (prev ?? []).map((r) => (r.id === row.id ? data.inbound! : r)));
   425→      toast({ title: lfmt(t.inbounds.toastUpdated, { tag: row.tag }) });
   426→    } catch {
   427→      toast({ title: t.inbounds.toastFail, variant: "destructive" });
   428→    }
   429→  };
   430→
   431→  const copyLink = async (row: InboundDTO) => {
   432→    const scheme = SCHEME[row.protocol] ?? row.protocol;
   433→    const link = `${scheme}://user@0.0.0.0:${row.port}?type=${row.transport}&security=${row.security}#${encodeURIComponent(row.tag)}`;
   434→    try {
   435→      await navigator.clipboard.writeText(link);
   436→      toast({ title: t.inbounds.toastCopied });
   437→    } catch {
   438→      toast({ title: t.inbounds.toastFail, variant: "destructive" });
   439→    }
   440→  };
   441→
   442→  /* ---------- render helpers ---------- */
   443→
   444→  const StatusText: Record<string, string> = {
   445→    online: t.inbounds.stOnline,
   446→    maintenance: t.inbounds.stMaint,
   447→    disabled: t.inbounds.stDisabled,
   448→  };
   449→
   450→  const KPI = ({ icon: Icon, value, label }: { icon: LucideIcon; value: string; label: string }) => (
   451→    <div className="glass hover-glow flex items-center gap-3.5 rounded-2xl px-4 py-3.5">
   452→      <span className="glass flex h-10 w-10 shrink-0 items-center justify-center rounded-xl">
   453→        <Icon className="h-4.5 w-4.5 text-primary" />
   454→      </span>
   455→      <div className="min-w-0">
   456→        <p className="truncate font-mono text-lg font-bold leading-tight text-silver tabular-nums">{value}</p>
   457→        <p className="truncate text-[11px] text-muted-foreground">{label}</p>
   458→      </div>
   459→    </div>
   460→  );
   461→
   462→  return (
   463→    <div className="space-y-6">
   464→      {/* ── header ─────────────────────────────────────────── */}
   465→      <div className="flex flex-wrap items-end justify-between gap-4">
   466→        <div>
   467→          <span className="inline-block rounded-full border border-primary/25 bg-primary/8 px-3 py-1 text-[10px] font-bold tracking-[0.18em] text-primary">
   468→            {t.inbounds.badge}
   469→          </span>
   470→          <h1 className="text-silver mt-3 text-2xl font-bold md:text-3xl">{t.nav.inbounds}</h1>
   471→          <p className="mt-1.5 max-w-xl text-sm leading-relaxed text-muted-foreground">
   472→            {t.inbounds.subtitle}
   473→          </p>
   474→        </div>
   475→        <Button
   476→          onClick={openCreate}
   477→          className="glow-cyan gap-2 rounded-full px-5 font-semibold shadow-lg shadow-primary/20"
   478→        >
   479→          <Plus className="h-4 w-4" />
   480→          {t.inbounds.new}
   481→        </Button>
   482→      </div>
   483→
   484→      {/* ── KPI strip ──────────────────────────────────────── */}
   485→      <div className="grid grid-cols-2 gap-3 md:gap-4 lg:grid-cols-4">
   486→        <KPI icon={RadioTower} value={String(counts.all)} label={t.inbounds.kpiTotal} />
   487→        <KPI icon={Wifi} value={String(counts.online)} label={t.inbounds.kpiOnline} />
   488→        <KPI icon={Users} value={String(totals.users)} label={t.inbounds.kpiUsers} />
   489→        <KPI icon={ArrowLeftRight} value={fmtTraffic(totals.traffic)} label={t.inbounds.kpiTraffic} />
   490→      </div>
   491→
   492→      {/* ── toolbar ────────────────────────────────────────── */}
   493→      <div className="flex flex-wrap items-center gap-3">
   494→        <div className="glass relative h-10 min-w-0 flex-1 sm:max-w-72">
   495→          <Search className="pointer-events-none absolute start-3 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-muted-foreground" />
   496→          <input
   497→            value={q}
   498→            onChange={(e) => setQ(e.target.value)}
   499→            placeholder={t.inbounds.searchPh}
   500→            aria-label={t.inbounds.searchPh}
   501→            className="h-full w-full bg-transparent ps-9 pe-3 text-sm outline-none placeholder:text-muted-foreground"
   502→          />
   503→        </div>
   504→        <div className="glass flex flex-wrap items-center rounded-full p-1" role="group" aria-label={t.nav.inbounds}>
   505→          {(
   506→            [
   507→              ["all", t.inbounds.fAll, counts.all],
   508→              ["online", t.inbounds.fOnline, counts.online],
   509→              ["maintenance", t.inbounds.fMaint, counts.maintenance],
   510→              ["disabled", t.inbounds.fDisabled, counts.disabled],
   511→            ] as const
   512→          ).map(([key, label, count]) => (
   513→            <button
   514→              key={key}
   515→              onClick={() => setStatusFilter(key)}
   516→              aria-pressed={statusFilter === key}
   517→              className={cn(
   518→                "flex h-8 items-center gap-1.5 rounded-full px-3 text-[12px] font-medium transition-colors",
   519→                statusFilter === key
   520→                  ? "bg-primary/15 text-primary ring-1 ring-primary/30"
   521→                  : "text-muted-foreground hover:text-foreground"
   522→              )}
   523→            >
   524→              {key !== "all" && (
   525→                <span className={cn("h-1.5 w-1.5 rounded-full", STATUS_DOT[key])} aria-hidden="true" />
   526→              )}
   527→              {label}
   528→              <span className="font-mono text-[10px] tabular-nums opacity-70">{count}</span>
   529→            </button>
   530→          ))}
   531→        </div>
   532→      </div>
   533→
   534→      {/* ── load error ─────────────────────────────────────── */}
   535→      {loadErr && rows === null && (
   536→        <div className="glass flex items-center gap-3 rounded-2xl border-chart-5/30 px-5 py-4 text-sm text-chart-5">
   537→          <RadioTower className="h-4 w-4 shrink-0" />
   538→          {t.inbounds.loadFail}
   539→          <Button variant="outline" size="sm" onClick={() => void load()} className="ms-auto rounded-full" aria-label="retry">
   540→            ↻
   541→          </Button>
   542→        </div>
   543→      )}
   544→
   545→      {/* ── cards / empty / loading ────────────────────────── */}
   546→      {rows === null ? (
   547→        <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
