   239→/** Same composition the core driver uses for its wire `proto` label. */
   240→function composeProto(protocol: string, transport: string, security: string): string {
   241→  const parts = [PROTO_LABEL[protocol] ?? protocol];
   242→  const tr = TRANSPORT_SHORT[transport] ?? transport.toUpperCase();
   243→  if (tr) parts.push(tr);
   244→  if (security === "tls") parts.push("TLS");
   245→  if (security === "reality") parts.push("REALITY");
   246→  return parts.join(" · ");
   247→}

   256→function hashSeed(s: string): number {
   257→  let h = 2166136261;
   258→  for (let i = 0; i < s.length; i++) {
   259→    h ^= s.charCodeAt(i);
   260→    h = Math.imul(h, 16777619);
   261→  }
   262→  return h >>> 0;
   263→}

   265→function mulberry32(a: number): () => number {
   266→  return () => {
   267→    a |= 0;
   268→    a = (a + 0x6d2b79f5) | 0;
   269→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
   270→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   271→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   272→  };
   273→}

   275→function Sparkline({
   276→  seed,
   277→  color,
   278→  className,
   279→}: {
   280→  seed: string;
   281→  color: string;
   282→  className?: string;
   283→}) {
   284→  const pts = useMemo(() => {
   285→    const rnd = mulberry32(hashSeed(seed));
   286→    const n = 26;
   287→    const arr: number[] = [];
   288→    let v = 0.45 + rnd() * 0.2;
   289→    for (let i = 0; i < n; i++) {
   290→      v += (rnd() - 0.48) * 0.16;
   291→      v = Math.min(0.95, Math.max(0.12, v));
   292→      arr.push(v);
   293→    }
   294→    return arr;
   295→  }, [seed]);
   296→  const W = 120;
   297→  const H = 40;
   298→  const step = W / (pts.length - 1);
   299→  const line = pts
   300→    .map((p, i) => `${(i * step).toFixed(1)},${(H - p * H).toFixed(1)}`)
   301→    .join(" ");
   302→  const gid = `sp-${hashSeed(seed).toString(36)}`;
   303→  return (
   304→    <svg
   305→      viewBox={`0 0 ${W} ${H}`}
   306→      className={cn("h-9 w-24 shrink-0", className)}
   307→      preserveAspectRatio="none"
   308→      aria-hidden="true"
   309→    >
   310→      <defs>
   311→        <linearGradient id={gid} x1="0" y1="0" x2="0" y2="1">
   312→          <stop offset="0%" style={{ stopColor: color, stopOpacity: 0.34 }} />
   313→          <stop offset="100%" style={{ stopColor: color, stopOpacity: 0 }} />
   314→        </linearGradient>
   315→      </defs>
   316→      <polygon points={`0,${H} ${line} ${W},${H}`} fill={`url(#${gid})`} />
   317→      <polyline
   318→        points={line}
   319→        fill="none"
   320→        style={{ stroke: color }}
   321→        strokeWidth="1.6"
   322→        strokeLinejoin="round"
   323→        strokeLinecap="round"
   324→      />
   325→    </svg>
   326→  );
   327→}

   331→function b64url(bytes: number): string {
   332→  const arr = new Uint8Array(bytes);
   333→  crypto.getRandomValues(arr);
   334→  let bin = "";
   335→  for (const b of arr) bin += String.fromCharCode(b);
   336→  return btoa(bin).replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
   337→}

   339→function genClientKey(protocol: string): string {
   340→  if (protocol === "vless" || protocol === "vmess") {
   341→    if (typeof crypto !== "undefined" && "randomUUID" in crypto) return crypto.randomUUID();
   342→    return b64url(16);

   347→function genSubId(): string {
   348→  const arr = new Uint8Array(8);

   371→function buildStream(form: FormState): Record<string, string | number | boolean> {
   372→  const out: Record<string, string | number | boolean> = {};
   373→  for (const [k, v] of Object.entries(form.stream)) {
   374→    if (v === "" || v === undefined || v === null) continue;
   375→    if (NUM_STREAM_KEYS.has(k)) out[k] = Number(v) || 0;
   376→    else if (BOOL_STREAM_KEYS.has(k)) out[k] = v === "true";
   377→    else out[k] = v;
   378→  }
   379→  if (form.security === "tls" || form.security === "reality") {
   380→    out.fingerprint = form.stream.fingerprint || "chrome";
   381→  }
   382→  if (form.transport === "grpc") out.multiMode = form.stream.multiMode === "true";
   383→  out.mux = form.mux;
   384→  const fb = form.fallbacks.filter((f) => f.dest.trim());
   385→  if (fb.length > 0) {
   386→    out.fallbacksJson = JSON.stringify(
   387→      fb.map((f) => ({
   388→        name: f.name.trim() || undefined,
   389→        dest: f.dest.trim(),
   390→        alpn: f.alpn.trim() || undefined,
   391→        path: f.path.trim() || undefined,
   392→      }))
   393→    );
   394→  }
   395→  return out;
   396→}

   425→function dtoToForm(dto: InboundDTO): FormState {
   426→  const stream: Record<string, string> = {};
   427→  for (const [k, v] of Object.entries(dto.stream ?? {})) {
   428→    stream[k] = BOOL_STREAM_KEYS.has(k) ? String(Boolean(v)) : String(v);
   429→  }
   430→  let fallbacks: FallbackForm[] = [];
   431→  try {
   432→    const raw = stream.fallbacksJson;
   433→    if (raw) {
   434→      const parsed = JSON.parse(raw) as Array<Record<string, string | undefined>>;
   435→      fallbacks = parsed.map((f) => ({
   436→        name: f.name ?? "",
   437→        dest: f.dest ?? "",
   438→        alpn: f.alpn ?? "",
   439→        path: f.path ?? "",
   440→      }));
   441→    }
   442→  } catch {
   443→    fallbacks = [];
   444→  }
   445→  return {
   446→    tag: dto.tag,
   447→    protocol: dto.protocol,
   448→    port: String(dto.port),
   449→    listen: dto.listen ?? "",
   450→    transport: dto.transport,
   451→    security: dto.security,
   452→    status: dto.status,
   453→    note: dto.note ?? "",
   454→    trafficLimitGb: dto.trafficLimitGb === null ? "" : String(dto.trafficLimitGb),
   455→    expiryDays: dto.expiryDays === null ? "" : String(dto.expiryDays),
   456→    clients: (dto.clients ?? []).map((c) => ({
   457→      email: c.email,
   458→      key: c.key,
   459→      subId: c.subId,
   460→      flow: c.flow ?? "",
   461→      ipLimit: String(c.ipLimit ?? 0),
   462→      trafficLimitGb: c.trafficLimitGb === null || c.trafficLimitGb === undefined ? "" : String(c.trafficLimitGb),
   463→      expiryDays: c.expiryDays === null || c.expiryDays === undefined ? "" : String(c.expiryDays),
   464→      enabled: c.enabled,
   465→    })),
   466→    sniffing: {
   467→      enabled: dto.sniffing?.enabled ?? false,
   468→      destOverride: dto.sniffing?.destOverride?.length
   469→        ? dto.sniffing.destOverride
   470→        : ["http", "tls"],
   471→      routeOnly: dto.sniffing?.routeOnly ?? false,
   472→    },
   473→    mux: dto.stream?.mux === true || dto.stream?.mux === "true",
   474→    fallbacks,
   475→    stream,
   476→  };
   477→}
